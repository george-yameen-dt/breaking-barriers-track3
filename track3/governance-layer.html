Added to the whiteboard: the guide requires three things on every call that the sketch leaves out: Bedrock Guardrails on model input and output, a person's approval before any change of state, and the Pause unit (withdraw a model, calls refused).

A Dynatrace Workflow starts the run

Today track3.py polls Dynatrace from the laptop until a problem opens. With a Workflow, the problem itself starts the agents in the cloud, and the laptop drops out. Not built yet.

// Task 1 · read_facts
import { execution } from "@dynatrace-sdk/automation-utils";
import { queryExecutionClient } from "@dynatrace-sdk/client-query";

export default async function () {
  const problem = (await execution()).event();   // the Davis problem that fired
  const res = await queryExecutionClient.queryExecute({ body: {
    query: `fetch logs, from: now()-10m
      | filter service.name == "mobile-env" and mobile.kind == "step"
      | sort timestamp desc | limit 1
      | fields mobile.policy, mobile.phones_dropped, mobile.phones_served`,
    requestTimeoutMilliseconds: 30000 } });
  const step = res.result?.records?.[0] ?? {};
  return { problem: problem.display_id,
           model_running: step["mobile.policy"],
           phones_dropped: step["mobile.phones_dropped"] };
}

// Task 2 · start_agents
import { result } from "@dynatrace-sdk/automation-utils";
import { credentialVaultClient } from "@dynatrace-sdk/client-classic-environment-v2";

const START_URL = "https://<api-id>.execute-api.us-east-2.amazonaws.com/prod/start"; // on the allowlist
const KEY_ID = "CREDENTIALS_VAULT-<id>";                                          // API Gateway key

export default async function () {
  const facts = await result("read_facts");
  const { token } = await credentialVaultClient.getCredentialsDetails({ id: KEY_ID });
  const res = await fetch(START_URL, {
    method: "POST",
    headers: { "content-type": "application/json", "x-api-key": token },
    body: JSON.stringify(facts),
  });
  if (!res.ok) throw new Error(`start failed: ${res.status}`);
  return res.json();   // { run_id, trace_id }; the agents keep running on AgentCore
}
